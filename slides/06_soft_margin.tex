      \begin{aligned}
      \xi_j(\langle w,p_j\rangle+\beta)&\geq 1-\eta_j,\\

      \end{aligned}

    \[
      \begin{aligned}
      \min_{w,\beta,\eta}\quad&
        \frac12\|w\|^2+C\sum_{j=1}^{m}\eta_j
      \end{aligned}
    \]

    \SplitBody
